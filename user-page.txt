    <script>
        // 页面加载时验证登录状态并初始化数据
        window.addEventListener('load', () => {
            const isLogin = localStorage.getItem('isLogin') === 'true';
            const userAccount = localStorage.getItem('userAccount');
            const userIdentity = localStorage.getItem('userIdentity');
            const userAvatarSrc = localStorage.getItem('userAvatar');
            const hostDatabase = JSON.parse(localStorage.getItem('hostDatabase') || '[]');
            const recruitDatabase = JSON.parse(localStorage.getItem('recruitDatabase') || '[]');

            // 未登录则跳转回首页
            if (!isLogin) {
                alert('请先登录');
                window.location.href = 'index.html';
                return;
            }

            // 填充基础用户信息
            document.getElementById('userName').textContent = userAccount || '匿名用户';
            const identityText = userIdentity === 'host' ? '主持人' : userIdentity === 'admin' ? '管理员' : '玩家';
            document.getElementById('userIdentity').textContent = `身份：${identityText}`;
            document.getElementById('userAccount').textContent = userAccount || '未绑定手机号';

            // 显示用户头像（支持更换同步）
            const avatarImg = document.getElementById('avatarImg');
            const avatarText = document.getElementById('avatarText');
            if (userAvatarSrc) {
                avatarImg.src = userAvatarSrc;
                avatarImg.style.display = 'block';
                avatarText.style.display = 'none';
            }

            // 更换头像功能（同步到主页面）
            document.getElementById('avatarInput').addEventListener('change', (e) => {
                const file = e.target.files[0];
                if (file) {
                    const reader = new FileReader();
                    reader.onload = (event) => {
                        const imgSrc = event.target.result;
                        avatarImg.src = imgSrc;
                        avatarImg.style.display = 'block';
                        avatarText.style.display = 'none';
                        localStorage.setItem('userAvatar', imgSrc); // 同步存储
                    };
                    reader.readAsDataURL(file);
                }
            });

            // 渲染我的预约/主持日程（状态联动核心逻辑）
            const myReservations = document.getElementById('myReservations');
            const reservationTitle = document.getElementById('reservationTitle');

            if (userIdentity === 'host') {
                // 主持人身份：显示主持日程 + 预约审核
                reservationTitle.textContent = '我的主持日程';
                const currentHost = hostDatabase.find(host => host.name === userAccount) || hostDatabase[0];
                
                let scheduleHtml = `<table class="schedule-table">
                    <thead>
                        <tr>
                            <<th>日期</</th>
                            <<th>时间段</</th>
                            <<th>当前状态</</th>
                            <<th>操作</</th>
                        </tr>
                    </thead>
                    <tbody>`;

                currentHost.schedule.forEach(item => {
                    // 从跑团数据库匹配对应活动
                    const matchedRecruit = recruitDatabase.find(r => r.time.includes(item.date));
                    const activityName = matchedRecruit ? matchedRecruit.title : '自定义活动';

                    scheduleHtml += `<tr>
                        <td>${item.date}</td>
                        <td>${item.time}</td>
                        <td class="status-${item.status === '空闲' ? 'free' : item.status === '预约中' ? 'reserving' : 'reserved'}">${item.status}</td>
                        <td>
                            ${item.status === '预约中' ? `
                                <button class="status-btn confirm" onclick="updateScheduleStatus('${item.date}', '${item.time}', '预约完成')">确认预约</button>
                                <button class="status-btn refuse" onclick="updateScheduleStatus('${item.date}', '${item.time}', '空闲')">取消预约</button>
                            ` : item.status === '空闲' ? `
                                <button class="status-btn confirm" onclick="updateScheduleStatus('${item.date}', '${item.time}', '预约中')">开启预约</button>
                            ` : `
                                <button class="status-btn secondary" onclick="updateScheduleStatus('${item.date}', '${item.time}', '空闲')">重置为空闲</button>
                            `}
                        </td>
                    </tr>`;
                });

                scheduleHtml += `</tbody></table>`;
                myReservations.innerHTML = scheduleHtml;

            } else {
                // 玩家/管理员身份：显示我的预约（关联跑团活动）
                reservationTitle.textContent = '我的预约';
                // 读取玩家预约记录（localStorage存储）
                const playerReservations = JSON.parse(localStorage.getItem('playerReservations') || '[]');
                
                if (playerReservations.length === 0) {
                    myReservations.innerHTML = '<p style="text-align:center; padding:20px;">暂无预约记录，快去首页预约跑团活动吧～</p>';
                } else {
                    let reservationHtml = '';
                    playerReservations.forEach(res => {
                        const matchedRecruit = recruitDatabase.find(r => r.id === res.recruitId);
                        reservationHtml += `<p>
                            ${matchedRecruit ? matchedRecruit.title : '已取消活动'} - ${res.date} ${res.time} 
                            <span class="opt-btn" onclick="cancelReservation(${res.recruitId})">取消预约</span>
                        </p>`;
                    });
                    myReservations.innerHTML = reservationHtml;
                }
            }

            // 身份专属功能区
            const identitySection = document.getElementById('identityExclusiveSection');
            if (userIdentity === 'host') {
                identitySection.innerHTML = `
                    <div class="section-title">主持人功能</div>
                    <div class="func-btn-group">
                        <button class="func-btn primary" onclick="window.location.href='release-recruit.html'">发布新跑团活动</button>
                        <button class="func-btn secondary">管理我的模组</button>
                    </div>
                    <div class="my-recruit">
                        <p><strong>已发布活动：</strong>${recruitDatabase.filter(r => r.host === userAccount).length}个</p>
                        <p><strong>待审核预约：</strong>${currentHost.schedule.filter(s => s.status === '预约中').length}个</p>
                        <p><strong>活动完成率：</strong>95%</p>
                    </div>
                `;
            } else if (userIdentity === 'admin') {
                identitySection.innerHTML = `
                    <div class="section-title">管理员功能</div>
                    <div class="func-btn-group">
                        <button class="func-btn primary" onclick="window.location.href='admin-backend.html'">进入管理后台</button>
                        <button class="func-btn secondary">审核活动/主持人</button>
                    </div>
                    <div class="admin-stat">
                        <div class="stat-card">
                            <div class="stat-num">${hostDatabase.length}</div>
                            <div class="stat-desc">平台主持人数量</div>
                        </div>
                        <div class="stat-card">
                            <div class="stat-num">${recruitDatabase.length}</div>
                            <div class="stat-desc">当前跑团活动数</div>
                            <div class="stat-desc">平台主持人数量</div>
                        </div>
                        <div class="stat-card">
                            <div class="stat-num">${JSON.parse(localStorage.getItem('playerReservations') || '[]').length}</div>
                            <div class="stat-desc">今日预约量</div>
                        </div>
                    </div>
                `;
            } else {
                // 玩家功能
                identitySection.innerHTML = `
                    <div class="section-title">玩家功能</div>
                    <div class="func-btn-group">
                        <button class="func-btn primary" onclick="window.location.href='index.html#recruit-area'">预约新活动</button>
                        <button class="func-btn secondary">查看活动评价</button>
                    </div>
                    <div class="my-info">
                        <p><strong>预约成功率：</strong>100%</p>
                        <p><strong>获得徽章：</strong>新手跑团者、资深调查员</p>
                    </div>
                `;
            }
        });

        // 全局函数：更新日程状态（同步到localStorage）
        window.updateScheduleStatus = function(date, time, newStatus) {
            const hostDatabase = JSON.parse(localStorage.getItem('hostDatabase') || '[]');
            const userAccount = localStorage.getItem('userAccount');
            const currentHost = hostDatabase.find(host => host.name === userAccount) || hostDatabase[0];
            
            // 更新对应日程状态
            const targetSchedule = currentHost.schedule.find(item => item.date === date && item.time === time);
            if (targetSchedule) {
                targetSchedule.status = newStatus;
                localStorage.setItem('hostDatabase', JSON.stringify(hostDatabase));
                alert(`状态已更新为：${newStatus}`);
                window.location.reload(); // 刷新页面
            }
        };

        // 全局函数：玩家取消预约
        window.cancelReservation = function(recruitId) {
            let playerReservations = JSON.parse(localStorage.getItem('playerReservations') || '[]');
            playerReservations = playerReservations.filter(res => res.recruitId !== recruitId);
            localStorage.setItem('playerReservations', JSON.stringify(playerReservations));
            
            // 同步更新主持人日程为空闲
            const recruitDatabase = JSON.parse(localStorage.getItem('recruitDatabase') || '[]');
            const matchedRecruit = recruitDatabase.find(r => r.id === recruitId);
            if (matchedRecruit) {
                const hostDatabase = JSON.parse(localStorage.getItem('hostDatabase') || '[]');
                const targetHost = hostDatabase.find(h => h.name === matchedRecruit.host);
                if (targetHost) {
                    const targetSchedule = targetHost.schedule.find(s => s.date === matchedRecruit.time.split(' ')[0]);
                    if (targetSchedule) {
                        targetSchedule.status = '空闲';
                        localStorage.setItem('hostDatabase', JSON.stringify(hostDatabase));
                    }
                }
            }

            alert('预约已取消');
            window.location.reload();
        };
    </script>
