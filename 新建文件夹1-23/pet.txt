    <script>
        // 初始化数据
        let currentCoin = parseInt(localStorage.getItem('currentCoin')) || 8765920;
        let characterAttrs = JSON.parse(localStorage.getItem('characterAttrs')) || {
            programming: 120,
            eloquence: 150,
            medical: 90,
            physical: 180,
            inspiration: 230,
            popularity: 2634
        };
        // 宠物数据：心情（初始60）+ 上次更新时间 + 增益状态
        let petData = JSON.parse(localStorage.getItem('petData')) || Array(6).fill().map(() => ({
            mood: 60,
            lastUpdateTime: Date.now(),
            hasBuff: false
        }));

        // 页面加载初始化
        window.onload = function() {
            // 更新属性/金币显示
            updateInfoPanel();
            // 检查宠物饥饿（每7天心情-15）
            checkPetHunger();
            // 更新所有宠物心情显示
            updateAllPetMood();
        };

        // 更新左侧属性/金币面板
        function updateInfoPanel() {
            document.getElementById('current-coin').innerText = currentCoin;
            for (const [key, value] of Object.entries(characterAttrs)) {
                document.getElementById(`attr-${key}`).innerText = value;
            }
        }

        // 检查宠物饥饿
        function checkPetHunger() {
            const now = Date.now();
            const weekMs = 7 * 24 * 60 * 60 * 1000; // 7天毫秒数
            petData.forEach(pet => {
                if (now - pet.lastUpdateTime >= weekMs) {
                    pet.mood = Math.max(pet.mood - 15, 0);
                    pet.lastUpdateTime = now;
                    pet.hasBuff = false; // 心情下降后重置增益
                }
            });
            localStorage.setItem('petData', JSON.stringify(petData));
        }

        // 宠物互动逻辑
        function interactPet(petId, type) {
            const pet = petData[petId];
            const card = document.querySelectorAll('.pet-card')[petId];
            const feedCost = parseInt(card.dataset.feedCost);
            // 高级灵宠互动数值翻倍
            const isAdvanced = card.querySelector('.pet-tag') !== null;
            let moodChange = isAdvanced ? {
                feed: 30,
                play: 40,
                clean: 25
            } : {
                feed: 20,
                play: 25,
                clean: 15
            }[type];

            // 喂食逻辑（扣除金币）
            if (type === 'feed') {
                if (currentCoin < feedCost) {
                    alert(`金币不足！喂食需要${feedCost}金币`);
                    return;
                }
                currentCoin -= feedCost;
                localStorage.setItem('currentCoin', currentCoin);
            }

            // 更新心情（上限100）
            pet.mood = Math.min(pet.mood + moodChange, 100);
            pet.lastUpdateTime = Date.now();
            localStorage.setItem('petData', JSON.stringify(petData));
            updatePetMood(petId);

            // 触发增益（心情≥80且未触发）
            if (pet.mood >= 80 && !pet.hasBuff) {
                const buffAttr = card.dataset.buffAttr;
                const buffVal = parseInt(card.dataset.buffVal);
                characterAttrs[buffAttr] += buffVal;
                localStorage.setItem('characterAttrs', JSON.stringify(characterAttrs));
                pet.hasBuff = true;
                localStorage.setItem('petData', JSON.stringify(petData));
                updateInfoPanel(); // 更新属性显示
                alert(`${card.querySelector('.pet-name').innerText}心情愉悦！为你带来${getAttrName(buffAttr)}+${buffVal}的增益`);
            }

            // 提示信息
            const actionText = type === 'feed' ? '喂食' : type === 'play' ? '玩耍' : '清洁';
            alert(`成功${actionText}${card.querySelector('.pet-name').innerText}！心情+${moodChange}（当前：${pet.mood}）`);
            // 喂食后更新金币显示
            if (type === 'feed') updateInfoPanel();
        }

        // 更新单只宠物心情
        function updatePetMood(petId) {
            document.querySelectorAll('.mood-val')[petId].innerText = petData[petId].mood;
        }

        // 更新所有宠物心情
        function updateAllPetMood() {
            petData.forEach((pet, idx) => {
                document.querySelectorAll('.mood-val')[idx].innerText = pet.mood;
            });
        }

        // 属性名称映射
        function getAttrName(key) {
            const map = {
                programming: '编程',
                eloquence: '口才',
                medical: '医术',
                physical: '体能',
                inspiration: '灵感',
                popularity: '人气'
            };
            return map[key] || key;
        }
    </script>
